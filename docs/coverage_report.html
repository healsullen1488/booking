   7      def __init__(self, users, hasher, tokens):
   8          # users - репозиторий; hasher.make(p), hasher.check(p, h); tokens.make(user)
   9          self.users = users
  10          self.hasher = hasher
  11          self.tokens = tokens

  13      def register(self, email, name, password):
  14          email = email.strip()
  15          name = name.strip()
  16          if "@" not in email or email.startswith("@"):
  17              raise Invalid("bad email")
  18          if not name:
  19              raise Invalid("name is empty")
  20          if len(password) < 6:
  21              raise Invalid("password is too short")
  22          if len(password.encode("utf-8")) > 72:
  23              raise Invalid("password is too long")
  24          if self.users.by_email(email):
  25              raise Conflict("email already used")
  26          # регистрация всегда создаёт EMPLOYEE
  27          u = self.users.add(email, name, self.hasher.make(password), consts.EMPLOYEE)
  28          self.users.commit()
  29          return u
  30  
  31      def login(self, email, password):
  32          u = self.users.by_email(email.strip())
  33          if u is None or not self.hasher.check(password, u.pass_hash):
  34              raise Unauthorized("wrong email or password")
  35          return self.tokens.make(u)
  36  
  37      def set_role(self, actor, user_id, role_id):
  38          if actor.role_id != consts.ADMIN:
  39              raise Forbidden("only admin can change roles")

   8  def is_overlap(a1, a2, b1, b2):
   9      # интервалы [a1, a2) и [b1, b2) пересекаются, если a1 < b2 и b1 < a2.
  10      # "встык" (a2 == b1) пересечением не считается
  11      return a1 < b2 and b1 < a2

  14  def check_time(start, end, now, tz):
  15      # проверки по порядку; первая неудачная даёт Invalid
  16      if start.tzinfo is None or end.tzinfo is None:
  17          raise Invalid("time must have timezone")
  18      if end <= start:
  19          raise Invalid("end must be after start")
  20      d = end - start
  21      if d < timedelta(minutes=consts.MIN_MIN):
  22          raise Invalid("booking is shorter than 15 minutes")
  23      if d > timedelta(minutes=consts.MAX_MIN):
  24          raise Invalid("booking is longer than 4 hours")
  25      if start < now:
  26          raise Invalid("start is in the past")
  27      a = start.astimezone(tz)
  28      b = end.astimezone(tz)
  29      if a.date() != b.date():
  30          # бронь до ровно 21:00 заканчивается в тот же день, так что полночь сюда не попадает
  31          raise Invalid("booking must be inside one day")
  32      if a.time() < time(consts.OPEN_H) or b.time() > time(consts.CLOSE_H):
  33          raise Invalid("work time is 09:00-21:00")

  48  def day_window(day, tz):
  49      # рабочее окно даты day (datetime.date) в часовом поясе tz
  50      x = datetime(day.year, day.month, day.day, consts.OPEN_H, tzinfo=tz)
  51      y = datetime(day.year, day.month, day.day, consts.CLOSE_H, tzinfo=tz)
  52      return x, y

  55  def free_slots(busy, day, tz, now, minutes=consts.MIN_MIN):
  56      # busy - список (start, end) занятых интервалов
  57      # возвращает список свободных (start, end) не короче minutes минут
  58      x, y = day_window(day, tz)
  59      if now.astimezone(tz) > x:
  60          x = now.astimezone(tz)
  61      need = timedelta(minutes=minutes)
  62      res = []
  63      cur = x
  64      for s, e in sorted(busy):
  65          s = s.astimezone(tz)
  66          e = e.astimezone(tz)
  67          if e <= cur:
  68              continue
  69          if s >= y:
  70              break
  71          if s > cur and s - cur >= need:
  72              res.append((cur, s))
  73          if e > cur:
  74              cur = e
  75      if cur < y and y - cur >= need:
  76          res.append((cur, y))
  77      return res
